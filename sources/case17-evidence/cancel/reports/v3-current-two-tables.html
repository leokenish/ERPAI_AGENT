4* 선택근로 등록 직후 취소 전용 / SAP_BASIS 700
5* 입력한 사번, 시작일, 종료일이 모두 일치하는 한 기간만 처리한다.
6* 두 테이블을 정확한 키로 직접 물리 삭제하고 한 번만 저장한다.
7* 결재 연결과 실적 상태는 삭제 차단 조건으로 사용하지 않는다.
8* 대휴 및 결재 테이블, 정산 이력과 지급 자료는 변경하지 않는다.
9* 설치: 전용 T-code ZHR_CW_DEL을 만들고 인사부에만 부여한다.
10* 인사마스터는 조회/삭제하지 않는다. 동시 작업 제한은 README 참조.
11*---------------------------------------------------------------------*

131*---------------------------------------------------------------------*
132* 정확한 등록/기간을 읽고 마감 및 정산 흔적을 검사한다.
133*---------------------------------------------------------------------*
134FORM read_and_check.
135 DATA: lv_count TYPE i,
136 lv_month TYPE zthr0160-zmonth.
137 CLEAR gv_error.
138 REFRESH: gt_0132, gt_0112.
139
140* 전용 인사부 트랜잭션 권한은 실행 진입 시 검사한다.
141* 인사마스터 및 정보형 9021의 권한/등록은 이번 검사 대상이 아니다.
142 SELECT * INTO TABLE gt_0132 FROM zthr0132
143 WHERE pernr = p_pernr AND begda = p_begda
144 AND endda = p_endda AND z52wktp = gc_wktp.
145 DESCRIBE TABLE gt_0132 LINES lv_count.
146 IF lv_count <> 1.
147 gv_error = 'ZTHR0132의 선택근로 기간이 정확히 1건이어야 합니다.'.
148 RETURN.
149 ENDIF.
150 READ TABLE gt_0132 INTO gs_0132 INDEX 1.
151* 근무일 월이 아닌 선택기간 종료월로 마감을 확인한다.
152 SELECT SINGLE zmonth INTO lv_month FROM zthr0160
153 WHERE zmonth = gv_month AND z52wktp = gc_wktp.
154 IF sy-subrc = 0.
155 gv_error = '해당 정산연월의 선택근로 마감이 끝나 삭제할 수 없습니다.'.
156 RETURN.
157 ENDIF.

159* 마감 표시가 없어도 과거/소급 정산 흔적이 있으면 삭제 금지.
160 SELECT COUNT( * ) INTO lv_count FROM zthr0113
161 WHERE pernr = p_pernr AND begda = p_begda
162 AND endda = p_endda AND z52wktp = gc_wktp.
163 IF lv_count > 0.
164 gv_error = '해당 선택기간에 정산 이력이 있어 삭제할 수 없습니다.'.
165 RETURN.
166 ENDIF.

168* 날짜 범위로 자르지 않는다. 기간 키에 속한 모든 실적을 읽는다.
169 SELECT * INTO TABLE gt_0112 FROM zthr0112
170 WHERE pernr = p_pernr AND begda = p_begda
171 AND endda = p_endda AND z52wktp = gc_wktp.
172 LOOP AT gt_0112 TRANSPORTING NO FIELDS
173 WHERE z52prsq <> gs_0132-z52prsq
174 OR ( zadmit_dt <> '00000000' AND zadmit_dt <> space ).
175 gv_error = '실적에 다른 근무주기 또는 정산일이 있어 삭제할 수 없습니다.'.
176 RETURN.
177 ENDLOOP.
178 SORT gt_0112 BY pernr datum begda endda z52wktp z52prsq seqnr.
179 DESCRIBE TABLE gt_0112 LINES gv_count.

182*---------------------------------------------------------------------*
183* 직원 잠금 + 이 프로그램 간 공통 테이블 잠금.
184* E_TABLE은 기존 월마감/입력 프로그램이 자동으로 따르는 잠금이 아니다.
185* 운영 전 기존 쓰기 경로와 같은 잠금을 사용하도록 연계해야 한다.
186*---------------------------------------------------------------------*
187FORM lock_target.
188 CALL FUNCTION 'BAPI_EMPLOYEE_ENQUEUE'
189 EXPORTING number = p_pernr
190 IMPORTING return = gs_return
191 EXCEPTIONS OTHERS = 1.
192 IF sy-subrc <> 0 OR gs_return-type CA 'AEX'.
193 gv_error = '직원이 다른 작업 중이어서 잠글 수 없습니다.'.
194 RETURN.
195 ENDIF.
196 gv_locked = 'X'.
197 PERFORM lock_table USING 'ZTHR0160'.
198 PERFORM lock_table USING 'ZTHR0113'.
199 PERFORM lock_table USING 'ZTHR0132'.
200 PERFORM lock_table USING 'ZTHR0112'.
201ENDFORM.
202
203FORM lock_table USING pv_tabname TYPE rstable-tabname.
204 IF gv_error IS NOT INITIAL.
205 RETURN.
206 ENDIF.
207 CALL FUNCTION 'ENQUEUE_E_TABLE'
208 EXPORTING tabname = pv_tabname varkey = space _scope = '1'
209 EXCEPTIONS foreign_lock = 1 system_failure = 2 OTHERS = 3.
210 IF sy-subrc <> 0.
211 CONCATENATE pv_tabname '잠금을 얻지 못했습니다. 다시 실행하세요.'
212 INTO gv_error SEPARATED BY space.
213 ELSE.
214 APPEND pv_tabname TO gt_locks.
215 ENDIF.
216ENDFORM.

218FORM recheck_target.
219 DATA: lt_before_0132 TYPE STANDARD TABLE OF zthr0132,
220 lt_before_0112 TYPE STANDARD TABLE OF zthr0112.
221 lt_before_0132[] = gt_0132[].
222 lt_before_0112[] = gt_0112[].
223 PERFORM read_and_check.
224 IF gv_error IS NOT INITIAL.
225 RETURN.
226 ENDIF.
227 IF lt_before_0132[] <> gt_0132[] OR
228 lt_before_0112[] <> gt_0112[].
229 gv_error = '확인 중 자료가 변경됐습니다. 다시 조회하여 실행하세요.'.
230 ENDIF.
231ENDFORM.

233*---------------------------------------------------------------------*
234* 중간 COMMIT 없이 선택기간과 실적만 직접 삭제한다.
235* 인사마스터는 조회하거나 삭제하지 않는다.
236* 두 삭제 중 하나라도 실패하면 호출부에서 전체 ROLLBACK한다.
237*---------------------------------------------------------------------*
238FORM delete_target.
239 DATA lv_count TYPE i.
240
241* 결재 연결 및 실적 상태와 관계없이 해당 기간 실적을 물리 삭제.
242 DELETE FROM zthr0112 WHERE pernr = p_pernr
243 AND begda = p_begda AND endda = p_endda
244 AND z52wktp = gc_wktp AND z52prsq = gs_0132-z52prsq.
245 gv_deleted = sy-dbcnt.
246 IF gv_deleted <> gv_count.
247 gv_error = '실적 삭제 건수가 달라 전체 취소합니다.'.
248 RETURN.
249 ENDIF.
250 DELETE FROM zthr0132 WHERE pernr = p_pernr
251 AND begda = p_begda AND endda = p_endda
252 AND z52wktp = gc_wktp AND z52prsq = gs_0132-z52prsq.
253 IF sy-dbcnt <> 1.
254 gv_error = '선택기간 삭제 건수가 달라 전체 취소합니다.'.
255 RETURN.
256 ENDIF.
257
258* 저장 직전 마감 표시를 한 번 더 확인한다.
259 SELECT COUNT( * ) INTO lv_count FROM zthr0160
260 WHERE zmonth = gv_month AND z52wktp = gc_wktp.
261 IF lv_count > 0.
262 gv_error = '처리 중 마감이 확인되어 전체 취소합니다.'.
263 RETURN.
264 ENDIF.
265 COMMIT WORK AND WAIT.
266 IF sy-subrc <> 0.
267 gv_error = '갱신 오류입니다. 시스템 담당자가 저장 결과를 확인해야 합니다.'.
268 ENDIF.
269ENDFORM.
270
271FORM unlock_target.
272 LOOP AT gt_locks INTO gv_tabname.
273 CALL FUNCTION 'DEQUEUE_E_TABLE'
274 EXPORTING tabname = gv_tabname varkey = space _scope = '1'.
275 ENDLOOP.
276 REFRESH gt_locks.
277 IF gv_locked = 'X'.
278 CALL FUNCTION 'BAPI_EMPLOYEE_DEQUEUE'
279 EXPORTING number = p_pernr
280 IMPORTING return = gs_return.
281 CLEAR gv_locked.
282 ENDIF.
283ENDFORM.
